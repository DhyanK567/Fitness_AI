Constant-Time HMAC-SHA256 Signature Verification

backend-nestjs-gateway/src/common/signature.ts

Standard string comparison (===) leaks execution time differences depending on how many characters match, exposing cryptographic keys to timing attacks. Syed engineered verifyHmacSignature using constant-time buffer comparison.

signature.ts · Timing-Attack Immunity 
import * as crypto from 'crypto';

export function verifyHmacSignature(
  payload: string | Buffer,
  receivedSignature: string,
  secret: string,
): boolean {
  if (!receivedSignature || !secret) {
    return false;
  }
  const computedHex = crypto
    .createHmac('sha256', secret)
    .update(payload)
    .digest('hex');

  const computedBuffer = Buffer.from(computedHex, 'utf8');
  const receivedBuffer = Buffer.from(receivedSignature, 'utf8');

  if (computedBuffer.length !== receivedBuffer.length) {
    return false;
  }
  // Constant-time comparison ensures zero timing leakage
  return crypto.timingSafeEqual(computedBuffer, receivedBuffer);
}

Idempotent ERP Payment Posting & Manual UPI/Cash Entry

erpnext-custom-app/wellness_studio/api/payment.py

Yashwanth engineered post_payment and record_manual to handle online webhooks and offline payments with strict tenant boundaries and idempotency key checks.

payment.py · Deduplication & Status Tracking 
@frappe.whitelist()
def post_payment(
    tenant: str,
    member: str,
    amount: float,
    payment_type: str = "Membership",
    idempotency_key: str | None = None,
    razorpay_order_id: str | None = None,
    razorpay_payment_id: str | None = None,
):
    # 1. Enforce tenant & member validity
    if not tenant or not frappe.db.exists("Studio Tenant", tenant):
        frappe.throw(_("A valid tenant is required"), frappe.ValidationError)
    if not member or not frappe.db.exists("WS Member", {"name": member, "tenant": tenant}):
        frappe.throw(_("Member not found"), frappe.DoesNotExistError)
    
    # 2. Check Idempotency Key - prevent duplicate ledger entry
    existing = frappe.db.get_value(
        "WS Payment Record", {"tenant": tenant, "idempotency_key": idempotency_key},
        ["name", "status"], as_dict=True,
    )
    if existing:
        return {"payment_id": existing.name, "status": existing.status, "duplicate": True}

    # 3. Create Verified Paid Record
    payment = frappe.get_doc({
        "doctype": "WS Payment Record", "tenant": tenant, "member": member,
        "payment_type": payment_type, "amount": amount, "currency": "INR",
        "razorpay_order_id": razorpay_order_id, "razorpay_payment_id": razorpay_payment_id,
        "idempotency_key": idempotency_key, "status": "Paid", "paid_on": frappe.utils.now_datetime(),
    })
    payment.insert(ignore_permissions=True)
    return {"payment_id": payment.name, "status": "Paid", "duplicate": False}

Tenant-Scoped Configuration & Dashboard Aggregation

NestJS Gateway Services

Both ConfigurationService and DashboardService strictly enforce tenant boundaries to prevent any cross-tenant data leaks during report generation.

configuration.service.ts
async getSettings(tenantId: string) {
  return this.prisma.tenantSetting.findMany({
    where: { tenantId },
  });
}
dashboard.service.ts
async getStats(tenantId: string) {
  const outboxCount = await this.prisma.outboxEvent.count({
    where: { tenantId },
  });
  return { tenantId, pendingEvents: outboxCount };
}
